Arquitectura que usaría: Monorepo pnpm + Turborepo

Separación radical de concerns. De 35k líneas hoy mezcladas en un solo paquete a 6 paquetes aislados. No Next 16.4 todo-en-uno con PPR disciplina obligatoria.

6 Paquetes / Aplicaciones Autónomas pnpm + Turborepo
📦 packages/core
Motor fiscal TS puro, 0 dependencias externas, 100% funciones puras (SBC, ISR, IMSS). Corre idéntico en browser, Cloudflare Worker y Node. Path directo a Rust/WASM para v2.
📦 packages/serializers
9 layouts aislados. Cada uno con test de importabilidad REAL en CONTPAQi 2024, sin mocks. El pipe | es el contrato fiscal innegociable.
📦 packages/db

🚀 apps/landing
Astro 4 + Tailwind. 100% estática, 0 JS en cliente, 30KB de peso total. No Next PPR para landing: es un impuesto de complejidad innecesario.

Remix o Next 15 App Router en Node runtime. Exclusivo para el despacho contable. Server actions para auth y RLS. Cero PPR disciplina.
⚡ apps/reader
Servicio minúsculo Go para lector NFC USB HID NTAG213. Emite evento directo a Supabase Realtime. Binario compilado local <5MB.

-- empresas: catálogo de cuentas contables parametrizable por cada despacho
CREATE TABLE empresas (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  rfc TEXT NOT NULL CHECK (char_length(rfc) BETWEEN 12 AND 13),
  nombre TEXT NOT NULL,
  catalogo_cuentas JSONB NOT NULL DEFAULT '{}', -- {"sueldo":"510-001", "extra":"..."}
  created_at TIMESTAMPTZ DEFAULT now()
);

-- trabajadores: datos fiscales mínimos indispensables
CREATE TABLE trabajadores (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  empresa_id UUID NOT NULL REFERENCES empresas(id),
  rfc TEXT NOT NULL, curp TEXT NOT NULL, nss TEXT NOT NULL,
  salario_diario NUMERIC(10,2) NOT NULL,
  sbc NUMERIC(10,2) NOT NULL,
  banco TEXT, cuenta TEXT,
  activo BOOLEAN DEFAULT TRUE,
  UNIQUE(empresa_id, rfc)
);

-- marcajes: INMUTABLE — sin UPDATE/DELETE — prueba plena Art. 805 LFT
CREATE TABLE marcajes (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  trabajador_id UUID NOT NULL REFERENCES trabajadores(id),
  timestamp_utc TIMESTAMPTZ NOT NULL DEFAULT now(),
  hash_sha256 TEXT NOT NULL, -- SHA256(trabajador_id + timestamp_utc + dispositivo_id)
  dispositivo_id TEXT NOT NULL,
  creado_en TIMESTAMPTZ DEFAULT now()
);

ALTER TABLE marcajes ENABLE ROW LEVEL SECURITY;
-- RLS: empresa_id vía trabajadores. NO hay política de UPDATE ni DELETE (solo INSERT/SELECT)
CREATE INDEX marcajes_brin ON marcajes USING BRIN (timestamp_utc);

-- periodos quincenales calculados
CREATE TABLE periodos (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  empresa_id UUID REFERENCES empresas(id),
  inicio DATE NOT NULL, fin DATE NOT NULL,
  estado TEXT CHECK (estado IN ('borrador', 'calculado')) DEFAULT 'borrador',
  total_neto NUMERIC(12,2), total_isr NUMERIC(12,2)
);

-- polizas: debe = haber, diferencia 0.00 estricta obligatoria
CREATE TABLE polizas (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  periodo_id UUID REFERENCES periodos(id),
  debe NUMERIC(12,2) NOT NULL,
  haber NUMERIC(12,2) NOT NULL,
  diferencia NUMERIC(12,2) GENERATED ALWAYS AS (debe - haber) STORED CHECK (diferencia = 0.00),
  movimientos JSONB NOT NULL -- [{cuenta, debe, haber, concepto}]
);

// Función pura determinista: corre en Web Worker <200ms para 100 empleados
export function calcularNominaQuincenal(e: EntradaNomina): SalidaNomina {
  // 1. Sueldo base
  const sueldo = e.sbc * e.dias_laborados;

  // 2. Horas extra — exención art 109 LISR
  // Doble: exento 50% hasta 5 UMA semanal, triple: 100% gravado
  const uma = getUMA(e.ejercicio);
  const topeExentoExtra = uma * 5;
  const extraDobleTotal = e.horas_extra_doble * (e.sbc / 8) * 2;
  const extraDobleExento = Math.min(extraDobleTotal * 0.5, topeExentoExtra);
  const extraDobleGravado = extraDobleTotal - extraDobleExento;
  const extraTripleTotal = e.horas_extra_triple * (e.sbc / 8) * 3; // 100% gravado

  // 3. Prima dominical — exenta hasta 1 SM art 71 LFT
  const primaTotal = e.prima_dominical_dias * e.sbc * 0.25;
  const sm = getSalarioMinimo(e.ejercicio);
  const primaExenta = Math.min(primaTotal, sm * e.prima_dominical_dias);
  const primaGravada = primaTotal - primaExenta;

  // 4. Base ISR art 96 — Tarifa 2026 (sin subsidio al empleo)
  const baseGravable = sueldo + extraDobleGravado + extraTripleTotal + primaGravada;
  const tarifa = getTarifaISR2026();
  const isr = tarifa.calcular(baseGravable);

  // 5. IMSS — 13 conceptos (Corrección de bug: guarderías NO suma a patronal)
  const imss = calcularIMSS13({ sbc: e.sbc, dias: e.dias_laborados, ejercicio: e.ejercicio });

  const neto = baseGravable - isr - imss.obrero.total + extraDobleExento + primaExenta;
  const debe = sueldo + extraDobleTotal + extraTripleTotal + primaTotal + imss.patronal.total;
  const haber = isr + imss.obrero.total + imss.patronal.total + neto;

  // Invariante contable obligatoria: debe == haber
  if (Math.abs(debe - haber) > 0.01) throw new Error("POLIZA_NO_CUADRA");

  return { sueldo, isr_retenido: isr, neto, debe, haber, ... };
}
